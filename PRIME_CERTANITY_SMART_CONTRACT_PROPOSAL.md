contract PRIMECertaintyOracle {
    using Counters for Counters.Counter;

    struct Proof {
        bytes32 proofHash;
        uint256 certaintyScore; // Always 10000 for valid proofs
        uint256 timestamp;
        address prover;
        bytes32 dataHash;
        bool isAbsolute;
    }
    
    // Mapping from data hash to proof
    mapping(bytes32 => Proof) private _proofs;
    
    // Mapping from proof ID to proof
    mapping(uint256 => Proof) private _proofsById;
    
    // Whitelisted provers (initially just PRIME network)
    mapping(address => bool) public whitelistedProvers;
    
    // Fee configuration
    uint256 public verificationFee = 0.01 ether;
    address public feeRecipient;
    
    // Events
    event ProofSubmitted(
        uint256 indexed proofId,
        bytes32 indexed dataHash,
        uint256 certaintyScore,
        uint256 timestamp
    );
    
    event SimulationDetected(
        bytes32 indexed dataHash,
        string reason,
        uint256 timestamp
    );
    
    constructor(address _feeRecipient) {
        feeRecipient = _feeRecipient;
        // Whitelist PRIME network addresses
        whitelistedProvers[msg.sender] = true;
    }
    
    // Submit a new certainty proof
    function submitProof(
        bytes32 dataHash,
        uint256 certaintyScore,
        bytes memory proofData
    ) external payable {
        require(whitelistedProvers[msg.sender], "PRIME: Not whitelisted");
        require(msg.value >= verificationFee, "PRIME: Insufficient fee");
        require(certaintyScore == 0 || certaintyScore == 10000, "PRIME: Invalid score");
        
        // Verify the proof (off-chain computation via PRIME network)
        bool isValid = _verifyProof(dataHash, proofData);
        require(isValid, "PRIME: Invalid proof");
        
        _proofIdCounter.increment();
        uint256 proofId = _proofIdCounter.current();
        
        Proof storage proof = Proof({
            proofHash: keccak256(proofData),
            certaintyScore: certaintyScore,
            timestamp: block.timestamp,
            prover: msg.sender,
            dataHash: dataHash,
            isAbsolute: certaintyScore == 10000
        });
        
        _proofs[dataHash] = proof;
        _proofsById[proofId] = proof;
        
        // Transfer fee to recipient
        payable(feeRecipient).transfer(verificationFee);
        
        emit ProofSubmitted(proofId, dataHash, certaintyScore, block.timestamp);
    }
    
    // Get proof by data hash
    function getProof(bytes32 dataHash) external view returns (Proof memory) {
        return _proofs[dataHash];
    }
    
    // Get proof by ID
    function getProofById(uint256 proofId) external view returns (Proof memory) {
        return _proofsById[proofId];
    }

        require(msg.sender == feeRecipient, "PRIME: Only fee recipient can whitelist");
        whitelistedProvers[prover] = true;
    }
    
    // Update verification fee

        require(msg.sender == feeRecipient, "PRIME: Only fee recipient can set fee");
        verificationFee = newFee;
    }
